onmouseover 
    document.getElementById("box")
    .onmouseover = function(){
        this.style.background = "yellow";
    };
Runs when mouse enters element.
onmouseout 
    document.getElementById("box")
    .onmouseout = function(){
        this.style.background = "white";
    };
Runs when mouse leaves element.
onkeydown 
    document.addEventListener("keydown", function(e){

    });
Detects when key is pressed.
onkeyup 
    document.addEventListener("keyup", function(){
        console.log("Key released");
    });
Detects when key is released.
onload

    };
Runs code when page fully loads.

    function validate(){

        if(name === ""){
            alert("Name required!");
            return false;
        }

        alert("Valid!");
    }

Full Example (Form + Event) 
    form onsubmit="return checkForm()">
        input id="name" placeholder="Name"
        button type="submit">Submit button
    form

    function checkForm(){
        let name = document.getElementById("name").value;

        if(name === ""){
            alert("Please enter name");
            return false;
        }

        document.getElementById("msg").innerHTML =
        "Hello " + name;

        return false;
    }
example: form, validation, alert, and DOM update.
Click Event (addEventListener) 
<button id="btn">Click Me</button>

<script>
document.getElementById("btn").addEventListener("click", function() {
    alert("Button Clicked!");
});
</script>
Button click triggers alert using addEventListener.
Mouse Over Event 
<div id="box">Hover me</div>

<script>
document.getElementById("box").addEventListener("mouseover", function() {
    console.log("Mouse is over the element");
});
</script>
Detects mouse entering the element.
Keyboard Event 
<script>
document.addEventListener("keydown", function(event) {
    console.log("Key pressed: " + event.key);
});
</script>
Tracks keyboard key presses.
Input Change Event 
<input id="input" type="text">

<script>
document.getElementById("input").addEventListener("change", function() {
    alert("Value changed!");
});
</script>
Runs when input value changes.
Form Submit Event 
<form id="form">
    <button type="submit">Submit</button>
</form>

<script>
document.getElementById("form").addEventListener("submit", function(e) {
    e.preventDefault();
    alert("Form Submitted!");
});
</script>
Prevents reload and handles form submit.
Image Click (Change Image) 
<img id="myImage" src="image1.jpg" width="100">

<script>
document.getElementById("myImage").addEventListener("click", function() {
    this.src = "image2.jpg";
});
</script>
Changes image when clicked.
